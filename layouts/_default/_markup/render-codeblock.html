{{- $result := transform.HighlightCodeBlock . -}} {{- /* Rename language-* classes to syntax-* to prevent 1Password prism.js interference */ -}} {{- /* Handle both single class and multiple classes: class="language-python" or class="highlight language-python" */ -}} {{- $wrapped := replaceRE `class="([^"]*\s)?language-([^"]+)"` `class="$1syntax-$2"` $result.Wrapped -}} {{- $wrapped = replaceRE `class="([^"]*\s)?lang-([^"]+)"` `class="$1syntax-$2"` $wrapped -}} {{- /* Also rename data-lang attributes */ -}} {{- $wrapped = replaceRE `data-lang="([^"]+)"` `data-syntax="$1"` $wrapped -}} {{- /* Hash a placeholder-free copy of the source (DOC-1357). A code block that embeds a shortcode still carries Hugo's internal HAHAHUGOSHORTCODEsHBHB placeholder at this point, and the counter is assigned per build — so hashing .Inner directly gives the copy button a different id on every deploy. Normalizing the placeholder away makes the id stable; .Ordinal still keeps it unique on the page. */ -}} {{- $hashSource := replaceRE `(?i)HAHAHUGOSHORTCODE\d+s\d+HBHB` "SHORTCODE" .Inner -}} {{- $uniqueID := printf "code-%s-%d" (sha1 $hashSource) .Ordinal -}}
