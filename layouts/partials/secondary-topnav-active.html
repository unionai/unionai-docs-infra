{{- /* Returns true if the secondary topnav is active for the current page+variant. A page's top-level section is "secondary-topnav-active" when its frontmatter has `secondary_topnav` set to a variant string that includes the current variant. secondary_topnav syntax (same as page variants): secondary_topnav: "+union -flyte" # active in union, inactive in flyte secondary_topnav: "+flyte +union" # active in both */ -}} {{- $topSection := false -}} {{- range .Site.Sections -}} {{- if eq $.Section .Section -}}{{- $topSection = . -}}{{- end -}} {{- end -}} {{- $active := false -}} {{- if and $topSection $topSection.Params.secondary_topnav -}} {{- $variant := .Site.Params.variant -}} {{- $rules := split $topSection.Params.secondary_topnav " " -}} {{- $isAllowed := in $rules (printf "+%s" $variant) -}} {{- $isExcluded := in $rules (printf "-%s" $variant) -}} {{- if and $isAllowed (not $isExcluded) -}} {{- $active = true -}} {{- end -}} {{- end -}} {{- return $active -}}
