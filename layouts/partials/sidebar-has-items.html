{{- /* Returns true if the sidebar will render any nav items for the current page+variant. Mirrors the iteration logic in partials/sidebar.html so the two stay in sync. Used by baseof.html to add a `no-sidebar` body class when the sidebar would otherwise render empty (e.g. top-level pages with no children, or a secondary-topnav root page where no child is active). */ -}} {{- $current := . -}} {{- $sidebarEnabled := .Params.sidebar_enabled | default true -}} {{- $hasItems := false -}} {{- if $sidebarEnabled -}} {{- $secondaryTopnavActive := partial "secondary-topnav-active.html" $current -}} {{- range .Site.Sections -}} {{- if eq $current.Section .Section -}} {{- if $secondaryTopnavActive -}} {{- $activeChild := false -}} {{- range .Pages.ByWeight -}} {{- if and (not $activeChild) (hasPrefix $current.RelPermalink .RelPermalink) -}} {{- $activeChild = . -}} {{- end -}} {{- end -}} {{- if and $activeChild $activeChild.IsSection -}} {{- range $activeChild.Pages.ByWeight -}} {{- $access := partial "page-allowed.html" . -}} {{- if $access.allowed -}} {{- $hasItems = true -}} {{- end -}} {{- end -}} {{- end -}} {{- else -}} {{- range .Pages.ByWeight -}} {{- $access := partial "page-allowed.html" . -}} {{- if $access.allowed -}} {{- $hasItems = true -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $hasItems -}}
