{{- /* Returns true if the sidebar will render any nav items for the current page+variant. Mirrors partials/sidebar.html's full-tree-nav (DOC-1334): the sidebar now always renders the WHOLE site tree -- every allowed `top_menu` section is a root -- so "has items" means "any allowed root exists for this variant", independent of the current page. Used by baseof.html to add a `no-sidebar` body class (pages with sidebar_enabled: false, or a variant with no allowed sections). */ -}} {{- $hasItems := false -}} {{- $sidebarEnabled := .Params.sidebar_enabled | default true -}} {{- if $sidebarEnabled -}} {{- range .Site.Sections.ByWeight -}} {{- $access := partial "page-allowed.html" . -}} {{- if and $access.allowed .Params.top_menu -}} {{- $hasItems = true -}} {{- end -}} {{- end -}} {{- end -}} {{- return $hasItems -}}
