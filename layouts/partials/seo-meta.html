{{- $description := "" -}} {{- $pageType := partial "page-type.html" . -}} {{- /* Check if this is an API reference page and provide better descriptions. Reference pages get an intent-distinct meta description (keyed to the symbol/page title) so they don't compete with concept/guide pages for the same intro terms in external search (DOC-1234). An explicit front-matter `description` always wins. */ -}} {{- if or (eq $pageType "reference") (.Description) -}} {{- if .Description -}} {{- $description = .Description -}} {{- else -}} {{- $description = printf "API reference for %s — generated documentation for the Flyte SDK and Union.ai Python API." .Title -}} {{- end -}} {{- else -}} {{- /* Extract content description from plain text (avoids re-rendering shortcodes) */ -}} {{- /* Hugo's .Plain strips block tags without inserting a separator, so the page H1 (= .Title) runs straight into the first paragraph ("Container imagesThe image…"). TrimSpace first (.Plain can lead with newlines), then drop a leading copy of the title before extracting the first sentence (DOC-1247). */ -}} {{- $body := .Plain | strings.TrimSpace -}} {{- if and .Title (hasPrefix $body .Title) -}} {{- $body = strings.TrimPrefix .Title $body | strings.TrimSpace -}} {{- end -}} {{- $sentences := findRE `[^.!?]+[.!?]` $body -}} {{- if gt (len $sentences) 0 -}} {{- $description = index $sentences 0 | replaceRE `\s+` " " | strings.TrimSpace -}} {{- else -}} {{- $description = .Summary | plainify -}} {{- end -}} {{- end -}} {{- /* Clean and normalize the description */ -}} {{- $description = $description | htmlUnescape -}} {{- /* Fix common malformed patterns */ -}} {{- $description = $description | replaceRE `Union\.ai ServerlessUnion\.ai` "Union.ai" -}} {{- $description = $description | replaceRE `Union\.ai ([A-Z][a-z]+)Union\.ai` "Union.ai" -}} {{- /* Fix missing spaces after periods */ -}} {{- $description = $description | replaceRE `\.([A-Z])` ". $1" -}} {{- /* Remove function signatures from API descriptions */ -}} {{- $description = $description | replaceRE `^def [^)]+\)\s*-?>?[^.]*\.?\s*` "" -}} {{- $description = $description | replaceRE `^Class Description\s+` "" -}} {{- /* Normalize whitespace and truncate */ -}} {{- $description = $description | replaceRE `\s+` " " | strings.TrimSpace -}} {{- if lt (len $description) 10 -}} {{- $description = "Union.ai documentation for AI workflow orchestration and machine learning pipelines." -}} {{- end -}} {{- /* The homepage is a redirect stub, so its front matter carries `description: Redirect to User guide` — which was surfacing as the site's meta description AND as WebSite.description in the JSON-LD. Prefer an explicit site-level description there (DOC-1287). */ -}} {{- if .IsHome -}} {{- with .Site.Params.site_description -}}{{- $description = . -}}{{- end -}} {{- end -}} {{- $variant := .Site.Params.variant -}} {{- $title := .Title | default .Site.Title -}} {{- /* Scheme+host for every fully-qualified URL below. The build runs with a path-only baseURL, so .Permalink and .Site.BaseURL are paths (see partials/site-host.html). */ -}} {{- $host := partial "site-host.html" . -}} {{- /* The canonical MUST be absolute (DOC-1287). A path-only canonical resolves against whatever host served the page, so every alternate host self-canonicalises and Google gets no signal to consolidate them. Measured 2026-08-11: the same v2 docs pages were independently indexed on www.union.ai, www-new.union.ai AND web-docs.union.ai, ~146 duplicates in a 1,000-row sample. That splits ranking signals between copies and works directly against DOC-1291 (noindex v1 to concentrate on v2) and DOC-1320 (one sitemap index). This value also feeds og:url, twitter:url, and the JSON-LD url/@id, so all five were relative. Guarded so a local build with an absolute baseURL is not double-prefixed. */ -}} {{- $canonical := .Permalink | absURL -}} {{- if not (hasPrefix $canonical "http") -}} {{- $canonical = printf "%s%s" $host $canonical -}} {{- end -}} {{- /* Same treatment for the site root, used by the homepage WebSite entity and the SearchAction target. schema.org wants absolute url/@id/target (DOC-1287). */ -}} {{- $siteURL := .Site.BaseURL -}} {{- if not (hasPrefix $siteURL "http") -}} {{- $siteURL = printf "%s%s" $host $siteURL -}} {{- end -}} {{- $product_full_name := index .Site.Params.key.product_full_name $variant -}} {{- $image := "" -}} {{- if eq $product_full_name "Flyte OSS" -}} {{- $image = printf "%s/images/icon-logo-flyte.svg" (strings.TrimSuffix "/" .Site.BaseURL) -}} {{- else -}} {{- $image = printf "%s/images/icon-logo.svg" (strings.TrimSuffix "/" .Site.BaseURL) -}} {{- end -}} {{- /* Crawlers (Google logo structured data, Open Graph, Twitter) require a FULLY-QUALIFIED absolute image URL, but .Site.BaseURL is a relative path (e.g. /docs/v2/union/) in production, so $image above is relative. $host (above) supplies the scheme+host (DOC-389). Leave it untouched if $image is already absolute (local dev may set an absolute baseURL). */ -}} {{- $logo := $image -}} {{- if not (hasPrefix $image "http") -}} {{- $logo = printf "%s%s" $host $image -}} {{- end -}} {{- if .Site.Params.twitter_site -}} {{- end -}} {{- if .Site.Params.twitter_creator -}} {{- end }} {{- if .IsHome -}} {{- /* Homepage: emit an Organization entity carrying the site `logo` — the schema.org field Google uses to associate a logo with the site in search results / the knowledge panel — alongside the WebSite entity, via @graph (DOC-389). The logo URL is absolute ($logo) and points at the real, variant-correct brand asset shipped in the theme. */ -}} {{- else -}} {{- end -}}
